using System;

// Tracks command recording only. RenderGraph RAW dependencies establish
// device execution order; release requires a separate last-use proof.
public sealed class NativeOwnerOrder154
{
    readonly int frame, ticket;
    int eyeLayers;
    bool world, prepared, selected, consumed, issued, failed, detached, released;

    public NativeOwnerOrder154(int engineFrame, int presentationTicket)
    {
        if(engineFrame<0||presentationTicket<=0)throw new ArgumentOutOfRangeException();
        frame=engineFrame;
        ticket=presentationTicket;
    }

    void RequireFrame(int engineFrame,int presentationTicket)
    {
        if(failed||engineFrame!=frame||presentationTicket!=ticket)
            throw new InvalidOperationException("154 stale or closed owner frame");
    }

    public void WorldIssued(int frame,int ticket)
    {
        RequireFrame(frame,ticket);
        if(world||prepared)throw new InvalidOperationException("154 duplicate world producer");
        world=issued=true;
    }

    public void EyeIssued(int frame,int ticket,int layer)
    {
        RequireFrame(frame,ticket);
        if(layer<0||layer>=5||prepared||(eyeLayers&(1<<layer))!=0)
            throw new InvalidOperationException("154 duplicate or invalid eye producer");
        eyeLayers|=1<<layer;
        issued=true;
    }

    public void PrepareIssued(int frame,int ticket)
    {
        RequireFrame(frame,ticket);
        if(!world||eyeLayers!=31||prepared)
            throw new InvalidOperationException("154 candidate pass before original producers");
        prepared=true;
    }

    public void SelectionIssued(int frame,int ticket)
    {
        RequireFrame(frame,ticket);
        if(!prepared||selected)throw new InvalidOperationException("154 selection before ordered candidate pass");
        selected=true;
    }

    public void ConsumeOnce(int frame,int ticket)
    {
        RequireFrame(frame,ticket);
        if(!selected||consumed)throw new InvalidOperationException("154 owner not produced or already transferred");
        consumed=true;
    }

    // Not a device barrier. After a partially recorded failure the existing
    // trial owner may release only after proof, never just because Abort ran.
    public void Abort(Action detach,Func<bool> lastUseConfirmed,Action release)
    {
        failed=true;
        if(!detached)
        {
            detach();
            detached=true;
        }
        if(!released&&(!issued||lastUseConfirmed()))
        {
            release();
            released=true;
        }
    }
}
