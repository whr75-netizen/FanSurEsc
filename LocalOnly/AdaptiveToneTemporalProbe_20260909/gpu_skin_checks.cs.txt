using System;
using System.IO;
using System.Text;
using System.Collections.Generic;
using UnityEngine;
using UnityEditor;

// Synthetic GPU contract checks. This file does not modify scenes or assets.
// Main task owns execution after shader compilation. No wall-time claims.
internal class CommandScript : IRunCommand
{
    const int Width = 64, Height = 64, SubjectId = 37;
    static readonly string[] KernelNames = { "Clear", "GatherGroups", "GatherHistogram",
        "SolveTones", "GatherHighlights", "SolveHighlights", "Resolve" };
    static readonly string[][] KernelBuffers = {
        new [] { "Groups", "Histogram", "Tiles", "TileHistogram", "Supports", "HighlightHistogram", "Status" },
        new [] { "Groups", "Status" },
        new [] { "Groups", "Histogram", "Tiles", "TileHistogram", "Status" },
        new [] { "Groups", "Histogram", "Tiles", "TileHistogram", "Supports", "Palette", "Solutions", "Status" },
        new [] { "Groups", "Palette", "Supports", "Solutions", "HighlightHistogram", "Status" },
        new [] { "Groups", "Histogram", "HighlightHistogram", "Solutions", "Status" },
        new [] { "Groups", "Palette", "Supports", "Solutions", "Status" }
    };

    [Serializable]
    public sealed class Evidence
    {
        public string name, graphicsDevice, graphicsApi;
        public string inputColorFormat = "RGBAHalf", metadataFormat = "RGBA32";
        public string lightingFormat = "RGBAHalf", outputFormat = "ARGBFloat";
        public string metricUnits = "Skin on: received illumination Y; otherwise Oklab L";
        // RunCommand lifts DTOs out of CommandScript: use fixture dimensions here.
        public int width = 64, height = 64;
        public bool skinLightingEnabled, flatSkin, passed;
        public float illuminationScale, skinAlbedoScale;
        public float relativeError = .10f, illuminationFloor = .02f;
        public uint globalStatus;
        public int skinGroupCount, skinGuardedGroups, skinMinK, skinMaxK;
        public int changedSkinSeeds, validSkinSeeds, invalidStatusGroups;
        public bool depthPreserved, emptySeedsPreserved, inputColorsPreserved;
        public bool metadataPreserved, lightingPreserved, skinUsesObservedRgb;
        public float meanSkinOutputY;
        public string[] failures;
        public int[] skinK;
        public uint[] status, groups;
        public float[] sourceColor, seedColor, outputColor, sourceMeta, seedMeta, sourceLighting;
        public float[] sourceColorAfter, seedColorAfter, sourceMetaAfter, seedMetaAfter, lightingAfter;
        public float[] solutions;
    }

    [Serializable]
    public sealed class Summary
    {
        public string outputDirectory;
        public string scope = "Synthetic compute dispatch/readback; not a rendered character or visual approval";
        public bool passed, dayOtherRgbEqual, nightOtherRgbEqual, pairedSeedAndMetaEqual;
        public bool broadSkinHasMultipleTones, nightSkinRemainsDarker;
        public bool albedoScalingPreservesSkinK, flatSkinHasOneTone;
        public float dayOtherMaxRgbDifference, nightOtherMaxRgbDifference;
        public float daySkinMeanY, nightSkinMeanY, nightDaySkinMeanRatio;
        public string[] cases, failures;
    }

    struct Rgb : IEquatable<Rgb>
    {
        public float r, g, b;
        public Rgb(Color c) { r = c.r; g = c.g; b = c.b; }
        public bool Equals(Rgb other) { return r == other.r && g == other.g && b == other.b; }
        public override bool Equals(object other) { return other is Rgb && Equals((Rgb)other); }
        public override int GetHashCode() { unchecked { return (r.GetHashCode() * 397 ^ g.GetHashCode()) * 397 ^ b.GetHashCode(); } }
    }

    public void Execute(ExecutionResult result)
    {
        if (!SystemInfo.supportsComputeShaders) throw new NotSupportedException("Compute shaders unavailable.");
        ComputeShader original = AssetDatabase.LoadAssetAtPath<ComputeShader>(
            "Assets/Rendering/CharacterPresentation/CharacterAdaptiveTones.compute");
        if (original == null) throw new InvalidOperationException("Adaptive tone ComputeShader asset missing.");
        string root = Path.GetFullPath(Path.Combine(Application.dataPath,
            "../LocalOnly/AdaptiveToneTemporalProbe_20260909"));
        string outputRoot = Path.Combine(root, "gpu_skin_checks_" + DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        Directory.CreateDirectory(outputRoot);
        ComputeShader shader = UnityEngine.Object.Instantiate(original);
        shader.hideFlags = HideFlags.HideAndDontSave;
        result.RegisterObjectCreation(shader);
        try
        {
            Evidence dayOff = RunCase(result, shader, outputRoot, "day_off", false, 1f, 1f, false);
            Evidence dayOn = RunCase(result, shader, outputRoot, "day_on", true, 1f, 1f, false);
            Evidence nightOff = RunCase(result, shader, outputRoot, "night_off", false, .12f, 1f, false);
            Evidence nightOn = RunCase(result, shader, outputRoot, "night_on", true, .12f, 1f, false);
            Evidence albedoOn = RunCase(result, shader, outputRoot, "day_albedo_x4_on", true, 1f, 4f, false);
            Evidence flatOn = RunCase(result, shader, outputRoot, "flat_skin_on", true, 1f, 1f, true);
            Evidence[] evidence = { dayOff, dayOn, nightOff, nightOn, albedoOn, flatOn };
            Summary summary = new Summary { outputDirectory = outputRoot };
            List<string> failures = new List<string>();
            List<string> caseNames = new List<string>();
            foreach (Evidence item in evidence)
            {
                caseNames.Add(item.name);
                if (!item.passed) failures.Add(item.name + ": " + string.Join("; ", item.failures));
            }
            summary.dayOtherMaxRgbDifference = OtherDifference(dayOff, dayOn);
            summary.nightOtherMaxRgbDifference = OtherDifference(nightOff, nightOn);
            summary.dayOtherRgbEqual = summary.dayOtherMaxRgbDifference == 0f;
            summary.nightOtherRgbEqual = summary.nightOtherMaxRgbDifference == 0f;
            summary.pairedSeedAndMetaEqual = PairInputsEqual(dayOff, dayOn) && PairInputsEqual(nightOff, nightOn);
            summary.broadSkinHasMultipleTones = dayOn.skinMaxK > 1 && nightOn.skinMaxK > 1;
            summary.daySkinMeanY = dayOn.meanSkinOutputY;
            summary.nightSkinMeanY = nightOn.meanSkinOutputY;
            summary.nightDaySkinMeanRatio = dayOn.meanSkinOutputY > 0f ? nightOn.meanSkinOutputY / dayOn.meanSkinOutputY : 0f;
            summary.nightSkinRemainsDarker = nightOn.meanSkinOutputY < dayOn.meanSkinOutputY && nightOn.meanSkinOutputY > 0f;
            summary.albedoScalingPreservesSkinK = Equal(dayOn.skinK, albedoOn.skinK);
            summary.flatSkinHasOneTone = flatOn.skinGroupCount > 0 && flatOn.skinMinK == 1 && flatOn.skinMaxK == 1;
            Check(summary.dayOtherRgbEqual, "Day other RGB changed when Skin lighting was enabled.", failures);
            Check(summary.nightOtherRgbEqual, "Night other RGB changed when Skin lighting was enabled.", failures);
            Check(summary.pairedSeedAndMetaEqual, "Paired off/on inputs differ.", failures);
            Check(summary.broadSkinHasMultipleTones, "Broad Skin gradient did not require multiple tones in day and night.", failures);
            Check(summary.nightSkinRemainsDarker, "Night Skin output is not positive and darker than day.", failures);
            Check(summary.albedoScalingPreservesSkinK, "Changing only Skin albedo x4 changed the sorted group K distribution.", failures);
            Check(summary.flatSkinHasOneTone, "Constant Skin illumination did not select K=1 in every Skin group.", failures);
            summary.cases = caseNames.ToArray(); summary.failures = failures.ToArray(); summary.passed = failures.Count == 0;
            WriteNew(Path.Combine(outputRoot, "summary.json"), JsonUtility.ToJson(summary, true));
            result.Log("Synthetic Skin GPU checks passed={0}; failures={1}; artifact={2}",
                summary.passed, failures.Count, Path.Combine(outputRoot, "summary.json"));
            foreach (string failure in failures) result.Log("FAILED: {0}", failure);
        }
        finally { result.DestroyObject(shader); }
    }

    static Evidence RunCase(ExecutionResult result, ComputeShader shader, string outputRoot,
        string name, bool enabled, float lightScale, float albedoScale, bool flatSkin)
    {
        Color[] colors = new Color[Width * Height], meta = new Color[Width * Height];
        Color[] seeds = new Color[Width * Height], seedMeta = new Color[Width * Height];
        Color[] lighting = new Color[Width * Height];
        for (int y = 0; y < Height; ++y)
        for (int x = 0; x < Width; ++x)
        {
            int index = y * Width + x;
            bool skin = y < Height / 2;
            float e = (skin && flatSkin ? .8f : .4f + .8f * x / (Width - 1)) * lightScale;
            Color albedo = skin ? new Color(.6f * albedoScale, .4f * albedoScale, .2f * albedoScale) : new Color(.04f, .012f, .003f);
            colors[index] = new Color(albedo.r * e, albedo.g * e, albedo.b * e, 3f + y * .015625f);
            meta[index] = new Color(SubjectId / 255f, (skin ? 4f : 3f) / 255f, 2f / 255f, 0f);
            lighting[index] = skin ? new Color(e, 0f, 0f, 1f) : Color.clear;
            if ((x & 1) != 0 && (y & 1) != 0) { seeds[index] = colors[index]; seedMeta[index] = meta[index]; }
        }
        List<UnityEngine.Object> objects = new List<UnityEngine.Object>();
        Dictionary<string, GraphicsBuffer> buffers = new Dictionary<string, GraphicsBuffer>();
        RenderTexture previousActive = RenderTexture.active, output = null;
        try
        {
            Texture2D source = Input(result, objects, TextureFormat.RGBAHalf, colors, name + " source");
            Texture2D nativeMeta = Input(result, objects, TextureFormat.RGBA32, meta, name + " native ID");
            Texture2D seed = Input(result, objects, TextureFormat.RGBAHalf, seeds, name + " seed");
            Texture2D sparseMeta = Input(result, objects, TextureFormat.RGBA32, seedMeta, name + " seed ID");
            Texture2D nativeLighting = Input(result, objects, TextureFormat.RGBAHalf, lighting, name + " lighting");
            Evidence data = new Evidence { name = name, skinLightingEnabled = enabled,
                illuminationScale = lightScale, skinAlbedoScale = albedoScale, flatSkin = flatSkin,
                graphicsDevice = SystemInfo.graphicsDeviceName, graphicsApi = SystemInfo.graphicsDeviceType.ToString(),
                sourceColor = Rows(source.GetPixels()), sourceMeta = Rows(nativeMeta.GetPixels()),
                seedColor = Rows(seed.GetPixels()), seedMeta = Rows(sparseMeta.GetPixels()),
                sourceLighting = Rows(nativeLighting.GetPixels()) };
            output = new RenderTexture(Width, Height, 0, RenderTextureFormat.ARGBFloat, RenderTextureReadWrite.Linear);
            output.name = name + " output"; output.hideFlags = HideFlags.HideAndDontSave;
            output.enableRandomWrite = true; output.filterMode = FilterMode.Point; output.wrapMode = TextureWrapMode.Clamp;
            objects.Add(output); result.RegisterObjectCreation(output);
            if (!output.Create()) throw new InvalidOperationException("Failed to create synthetic GPU output.");
            AddBuffer(buffers, "Groups", 160 * 8); AddBuffer(buffers, "Histogram", 160 * 32 * 6);
            AddBuffer(buffers, "Tiles", 160 * 16 * 3); AddBuffer(buffers, "TileHistogram", 160 * 16 * 32);
            AddBuffer(buffers, "Supports", 160 * 2); AddBuffer(buffers, "HighlightHistogram", 160 * 32 * 5);
            AddBuffer(buffers, "Status", 161); AddBuffer(buffers, "Palette", 160 * 32, 16); AddBuffer(buffers, "Solutions", 160 * 3, 16);
            shader.SetInt("_ToneWidth", Width); shader.SetInt("_ToneHeight", Height); shader.SetInt("_ToneSubjectId", SubjectId);
            shader.SetFloat("_ToneError", .06f); shader.SetFloat("_ToneChromaError", .06f);
            shader.SetFloat("_ToneContrastRetention", .65f); shader.SetFloat("_ToneMinimumContrast", .025f);
            shader.SetFloat("_ToneHighlightMinimumContrast", .025f);
            shader.SetInt("_ToneHairLightingEnabled", 0);
            shader.SetInt("_ToneSkinLightingEnabled", enabled ? 1 : 0);
            shader.SetFloat("_ToneSkinRelativeError", .10f); shader.SetFloat("_ToneSkinIlluminationFloor", .02f);
            for (int index = 0; index < KernelNames.Length; ++index)
            {
                int kernel = shader.FindKernel(KernelNames[index]);
                foreach (string buffer in KernelBuffers[index]) shader.SetBuffer(kernel, "_Tone" + buffer, buffers[buffer]);
                if (index >= 1 && index <= 5) shader.SetTexture(kernel, "_ToneSourceColor", source);
                if (index != 0) shader.SetTexture(kernel, "_ToneSourceLighting", nativeLighting);
                if (index == 1 || index == 2 || index == 4 || index == 6) shader.SetTexture(kernel, "_ToneSourceMeta", nativeMeta);
                if (index == 6)
                {
                    shader.SetTexture(kernel, "_ToneCellColor", seed); shader.SetTexture(kernel, "_ToneCellMeta", sparseMeta);
                    shader.SetTexture(kernel, "_ToneOutput", output);
                }
                int groupsX = index == 0 ? 1280 : (index == 3 || index == 5 ? 160 : (Width + 7) / 8);
                int groupsY = index == 0 || index == 3 || index == 5 ? 1 : (Height + 7) / 8;
                shader.Dispatch(kernel, groupsX, groupsY, 1);
            }
            Texture2D readback = new Texture2D(Width, Height, TextureFormat.RGBAFloat, false, true);
            readback.hideFlags = HideFlags.HideAndDontSave; objects.Add(readback); result.RegisterObjectCreation(readback);
            RenderTexture.active = output; readback.ReadPixels(new Rect(0, 0, Width, Height), 0, 0, false); readback.Apply(false, false);
            RenderTexture.active = previousActive;
            data.outputColor = Rows(readback.GetPixels());
            data.sourceColorAfter = Rows(source.GetPixels()); data.seedColorAfter = Rows(seed.GetPixels());
            data.sourceMetaAfter = Rows(nativeMeta.GetPixels()); data.seedMetaAfter = Rows(sparseMeta.GetPixels());
            data.lightingAfter = Rows(nativeLighting.GetPixels());
            data.status = ReadUInt(buffers["Status"]); data.groups = ReadUInt(buffers["Groups"]);
            Vector4[] solutions = new Vector4[buffers["Solutions"].count]; buffers["Solutions"].GetData(solutions);
            data.solutions = VectorRows(solutions);
            Analyze(data);
            WriteNew(Path.Combine(outputRoot, name + ".json"), JsonUtility.ToJson(data));
            result.Log("Synthetic case={0}, pass={1}, global={2}, Skin groups={3}, K={4}..{5}, meanY={6}",
                name, data.passed, data.globalStatus, data.skinGroupCount, data.skinMinK, data.skinMaxK, data.meanSkinOutputY);
            return data;
        }
        finally
        {
            RenderTexture.active = previousActive;
            foreach (GraphicsBuffer buffer in buffers.Values) buffer.Dispose();
            if (output != null) output.Release();
            for (int index = objects.Count - 1; index >= 0; --index) result.DestroyObject(objects[index]);
        }
    }

    static void Analyze(Evidence data)
    {
        List<string> failures = new List<string>(); List<int> skinK = new List<int>();
        data.globalStatus = data.status[160];
        for (int group = 0; group < 160; ++group)
        {
            uint key = data.groups[group * 8]; if (key == 0) continue;
            if ((data.status[group] & 1u) == 0u) ++data.invalidStatusGroups;
            if ((key >> 16) != 5u) continue;
            ++data.skinGroupCount; if ((data.status[group] & 2u) != 0u) ++data.skinGuardedGroups;
            skinK.Add(Mathf.RoundToInt(data.solutions[group * 12]));
        }
        skinK.Sort(); data.skinK = skinK.ToArray();
        data.skinMinK = skinK.Count > 0 ? skinK[0] : 0; data.skinMaxK = skinK.Count > 0 ? skinK[skinK.Count - 1] : 0;
        data.inputColorsPreserved = Equal(data.sourceColor, data.sourceColorAfter) && Equal(data.seedColor, data.seedColorAfter);
        data.metadataPreserved = Equal(data.sourceMeta, data.sourceMetaAfter) && Equal(data.seedMeta, data.seedMetaAfter);
        data.lightingPreserved = Equal(data.sourceLighting, data.lightingAfter);
        data.depthPreserved = data.emptySeedsPreserved = data.skinUsesObservedRgb = true;
        HashSet<Rgb> observedSkin = new HashSet<Rgb>();
        for (int pixel = 0; pixel < Width * Height; ++pixel)
            if (Mathf.RoundToInt(data.sourceMeta[pixel * 4 + 1] * 255f) == 4) observedSkin.Add(new Rgb(ColorAt(data.sourceColor, pixel)));
        double sumY = 0;
        for (int pixel = 0; pixel < Width * Height; ++pixel)
        {
            Color seed = ColorAt(data.seedColor, pixel), output = ColorAt(data.outputColor, pixel);
            if (output.a != seed.a) data.depthPreserved = false;
            if (seed.a <= 0f && !SameColor(seed, output)) data.emptySeedsPreserved = false;
            if (seed.a <= 0f || Mathf.RoundToInt(data.seedMeta[pixel * 4 + 1] * 255f) != 4) continue;
            ++data.validSkinSeeds;
            if (!observedSkin.Contains(new Rgb(output))) data.skinUsesObservedRgb = false;
            if (!new Rgb(seed).Equals(new Rgb(output))) ++data.changedSkinSeeds;
            sumY += .2126 * output.r + .7152 * output.g + .0722 * output.b;
        }
        data.meanSkinOutputY = data.validSkinSeeds > 0 ? (float)(sumY / data.validSkinSeeds) : 0f;
        Check(data.globalStatus == 0u, "Global status is nonzero.", failures);
        Check(data.invalidStatusGroups == 0, "At least one active group is not READY.", failures);
        Check(data.skinGroupCount > 0 && data.validSkinSeeds > 0, "Skin groups or sparse seeds missing.", failures);
        Check(data.depthPreserved, "Output changed sparse depth/alpha.", failures);
        Check(data.emptySeedsPreserved, "Output changed empty seeds.", failures);
        Check(data.inputColorsPreserved, "Source or seed color texture was modified.", failures);
        Check(data.metadataPreserved, "Native or seed metadata was modified.", failures);
        Check(data.lightingPreserved, "Native illumination texture was modified.", failures);
        Check(data.skinUsesObservedRgb, "Skin output contains RGB absent from observed native Skin pixels.", failures);
        data.failures = failures.ToArray(); data.passed = failures.Count == 0;
    }

    static float OtherDifference(Evidence a, Evidence b)
    {
        float maximum = 0f;
        for (int pixel = 0; pixel < Width * Height; ++pixel)
        {
            int offset = pixel * 4;
            if (a.seedColor[offset + 3] <= 0f || Mathf.RoundToInt(a.seedMeta[offset + 1] * 255f) != 3) continue;
            for (int channel = 0; channel < 3; ++channel) maximum = Mathf.Max(maximum, Mathf.Abs(a.outputColor[offset + channel] - b.outputColor[offset + channel]));
        }
        return maximum;
    }
    static bool PairInputsEqual(Evidence a, Evidence b)
    { return Equal(a.sourceColor, b.sourceColor) && Equal(a.seedColor, b.seedColor) && Equal(a.sourceMeta, b.sourceMeta) && Equal(a.seedMeta, b.seedMeta) && Equal(a.sourceLighting, b.sourceLighting); }
    static bool Equal(float[] a, float[] b) { if (a.Length != b.Length) return false; for (int i = 0; i < a.Length; ++i) if (a[i] != b[i]) return false; return true; }
    static bool Equal(int[] a, int[] b) { if (a.Length != b.Length) return false; for (int i = 0; i < a.Length; ++i) if (a[i] != b[i]) return false; return true; }
    static bool SameColor(Color a, Color b) { return a.r == b.r && a.g == b.g && a.b == b.b && a.a == b.a; }
    static Color ColorAt(float[] a, int pixel) { int i = pixel * 4; return new Color(a[i], a[i + 1], a[i + 2], a[i + 3]); }
    static void Check(bool condition, string failure, List<string> failures) { if (!condition) failures.Add(failure); }
    static void WriteNew(string path, string json)
    { using (FileStream file = new FileStream(path, FileMode.CreateNew, FileAccess.Write, FileShare.None)) using (StreamWriter writer = new StreamWriter(file, new UTF8Encoding(false))) writer.Write(json); }
    static Texture2D Input(ExecutionResult result, List<UnityEngine.Object> objects, TextureFormat format, Color[] values, string name)
    {
        Texture2D texture = new Texture2D(Width, Height, format, false, true);
        texture.name = name; texture.hideFlags = HideFlags.HideAndDontSave; texture.filterMode = FilterMode.Point; texture.wrapMode = TextureWrapMode.Clamp;
        objects.Add(texture); result.RegisterObjectCreation(texture); texture.SetPixels(values); texture.Apply(false, false); return texture;
    }
    static void AddBuffer(Dictionary<string, GraphicsBuffer> buffers, string name, int count, int stride = 4)
    { buffers.Add(name, new GraphicsBuffer(GraphicsBuffer.Target.Structured, count, stride)); }
    static uint[] ReadUInt(GraphicsBuffer buffer) { uint[] data = new uint[buffer.count]; buffer.GetData(data); return data; }
    static float[] Rows(Color[] values)
    { float[] rows = new float[values.Length * 4]; for (int i = 0; i < values.Length; ++i) { rows[i * 4] = values[i].r; rows[i * 4 + 1] = values[i].g; rows[i * 4 + 2] = values[i].b; rows[i * 4 + 3] = values[i].a; } return rows; }
    static float[] VectorRows(Vector4[] values)
    { float[] rows = new float[values.Length * 4]; for (int i = 0; i < values.Length; ++i) { rows[i * 4] = values[i].x; rows[i * 4 + 1] = values[i].y; rows[i * 4 + 2] = values[i].z; rows[i * 4 + 3] = values[i].w; } return rows; }
}
